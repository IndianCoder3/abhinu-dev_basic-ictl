========================================================================
ABHINU.DEV BASIC ICTL - INTERPRETER COMMAND REFERENCE
========================================================================

COMPLETE LIST OF ALL SUPPORTED COMMANDS AND FEATURES:

1. TERMINAL COMMANDS
============================

Terminal.Echo(<expression>) 
  - Output text to console
  - Parameters: Any expression (string, variable, math result)
  - Example: Terminal.Echo("Hello, World!")
  - Example: Terminal.Echo("Count: " + Variables.X)

Terminal.Ask(<prompt>)
  - Get user input from console
  - Parameters: A string prompt
  - Returns: User input as string
  - Example: Variables.Name = Terminal.Ask("Enter your name: ")

Terminal.Style(<style_name>)
  - Set terminal output color/style
  - Available styles:
    * "red" - Red text
    * "green" - Green text
    * "blue" - Blue text
    * "yellow" - Yellow text
    * "cyan" - Cyan text
    * "magenta" - Magenta text
    * "bold" - Bold text
    * "reset" - Reset to default
  - Example: Terminal.Style("green")


2. VARIABLE COMMANDS
============================

Variables.New(<variable_name>)
  - Create/declare a new variable
  - Parameters: Unquoted identifier name
  - Example: Variables.New(MyVar)

Variables.<variable_name> = <expression>
  - Assign value to variable
  - Can assign: strings, numbers, math results, user input
  - Example: Variables.Counter = 42
  - Example: Variables.Name = "Alice"
  - Example: Variables.Result = Math.Eval(10 + 5)
  - Example: Variables.Input = Terminal.Ask("Enter value: ")

Variables.<variable_name>
  - Reference/read variable value
  - Use in string concatenation or expressions
  - Example: Terminal.Echo(Variables.Counter)
  - Example: Terminal.Echo("Hello " + Variables.Name)


3. MATH OPERATIONS
============================

Math.Eval(<arithmetic_expression>)
  - Evaluate mathematical expressions
  - Supported operators: +, -, *, /, ( )
  - Supports: integers, floats, negative numbers
  - Example: Math.Eval(10 + 5)
  - Example: Math.Eval((2 + 3) * 4)
  - Example: Math.Eval(3.5 + 2.5)
  - Example: Math.Eval(-10 + 5)
  - Example: Math.Eval(Variables.X * 2)

Math.Compare(<value_a>, <operator>, <value_b>)
  - Compare two numeric values
  - Returns: Boolean (true/false)
  - Operators: ">", "<", ">=", "<=", "==", "!="
  - Example: Math.Compare(10, ">", 5)
  - Example: Math.Compare(Variables.X, "<=", 100)

Math.Random(<min>, <max>)
  - Generate random number between min and max (inclusive)
  - Parameters: Two integers (min and max)
  - Returns: Random integer within range
  - Example: Math.Random(1, 100)
  - Example: Variables.Roll = Math.Random(1, 6)
  - Example: Math.Random(0, 1) for coin flip simulation


4. TIME OPERATIONS
============================

Time.Current(<format>)
  - Get current system time with specified format
  - Parameter: Format string (e.g., "hh:mm:ss tt", "yyyy-MM-dd", "HH:mm:ss")
  - Returns: Current time as formatted string
  - Common formats:
    * "hh:mm:ss tt" - 12-hour time with AM/PM (e.g., 03:45:30 PM)
    * "HH:mm:ss" - 24-hour time (e.g., 15:45:30)
    * "yyyy-MM-dd" - Date (e.g., 2025-03-13)
    * "yyyy-MM-dd HH:mm:ss" - Date and time
  - Example: Terminal.Echo(Time.Current("hh:mm:ss tt"))
  - Example: Variables.StartTime = Time.Current("HH:mm:ss")

Time.Wait(<milliseconds>)
  - Pause program execution for specified time
  - Parameter: Integer milliseconds (1000 ms = 1 second)
  - Example: Time.Wait(1000) # Wait 1 second
  - Example: Time.Wait(500) # Wait half second
  - Example: Program.Loop(5) { Terminal.Echo(".") ; Time.Wait(500) }


5. DATA COMPARISON
============================

Data.Compare(<value_a>, <value_b>)
  - Compare two values for equality (string or number)
  - Returns: Boolean (true/false)
  - Compares as strings
  - Example: Data.Compare("hello", "hello")
  - Example: Data.Compare(Variables.Input, "yes")


6. CONTROL FLOW - CONDITIONALS
============================

Program.If(<condition>) { ... }
  - Execute block if condition is true
  - Condition can be:
    * Data.Compare(...)
    * Math.Compare(...)
    * Boolean literal (True/False)
    * Result of any comparison
    * Program.Not(...)
  - Example:
    Program.If(Math.Compare(X, ">", 5)) {
        Terminal.Echo("X is greater than 5")
    }
  - Supports nesting:
    Program.If(condition1) {
        Program.If(condition2) {
            ...
        }
    }

Program.Else { ... }
  - Execute block if preceding Program.If condition was false
  - Must follow a Program.If block
  - Only one Program.Else per Program.If
  - Example:
    Program.If(Math.Compare(X, ">", 5)) {
        Terminal.Echo("X is greater than 5")
    }
    Program.Else {
        Terminal.Echo("X is 5 or less")
    }

Program.Not(<condition>)
  - Flip/negate a boolean condition
  - Returns opposite of condition (true becomes false, false becomes true)
  - Example: Program.Not(Math.Compare(1, "==", 1)) results in false (because Math.Compare is true)
  - Example: Program.If(Program.Not(Data.Compare(name, "admin"))) { ... }
  - Useful for inverting comparisons without complex logic


7. LOOPS - COUNTED LOOP
============================

Program.Loop(<count>) { ... }
  - Execute block N number of times
  - Parameter: Integer count
  - Example:
    Program.Loop(5) {
        Terminal.Echo("iteration")
    }
  - Supports loop control (Break/Continue) inside


8. LOOPS - INFINITE LOOP
============================

Program.ForeverLoop { ... }
  - Execute block infinitely until break
  - Must include Program.BreakLoop to exit
  - Example:
    Program.ForeverLoop {
        Variables.Counter = Math.Eval(Variables.Counter + 1)
        Program.If(Math.Compare(Variables.Counter, ">", 10)) {
            Program.BreakLoop
        }
    }


9. LOOP CONTROL STATEMENTS
============================

Program.BreakLoop
  - Exit current loop immediately
  - Use inside Program.Loop or Program.ForeverLoop
  - Example:
    Program.ForeverLoop {
        ...
        Program.BreakLoop
    }

Program.Continue
  - Skip to next loop iteration
  - Use inside Program.Loop or Program.ForeverLoop
  - Example:
    Program.Loop(10) {
        Program.If(Data.Compare(i, 5)) {
            Program.Continue
        }
        Terminal.Echo(i)
    }


10. MAIN PROGRAM BLOCK
============================

Program.Main { ... }
  - Main entry point of program
  - All code must be inside Program.Main block
  - Program execution starts here
  - Example:
    Program.Main {
        Terminal.Echo("Hello!")
    }


11. DATA TYPES
============================

String Literals
  - Enclosed in double quotes: "text"
  - Example: "Hello World"
  - Example: "42"

Integer Literals
  - Whole numbers with optional negative sign
  - Example: 42
  - Example: -5
  - Example: 0

Float Literals
  - Decimal numbers with optional negative sign
  - Example: 3.14
  - Example: -2.5
  - Example: 0.0

Boolean Literals
  - True (capitalized)
  - False (capitalized)
  - Example: Program.If(True) { ... }


12. OPERATORS
============================

String Concatenation
  - Operator: +
  - Joins strings and values together
  - Example: "Hello" + Variables.Name
  - Example: "Sum: " + Math.Eval(5 + 3)

Mathematical Operators
  - Addition: +
  - Subtraction: -
  - Multiplication: *
  - Division: /
  - Grouping: ( )
  - Used inside Math.Eval()

Comparison Operators (in Math.Compare)
  - Greater than: ">"
  - Less than: "<"
  - Greater or equal: ">="
  - Less or equal: "<="
  - Equal: "=="
  - Not equal: "!="


13. COMMENTS
============================

Single-line comments
  - Start with # or //
  - Example: # This is a comment
  - Example: // This is also a comment


14. SCOPE RULES
============================

- Variables are global (accessible anywhere after declaration)
- Variables persist across all blocks and loops
- Functions/blocks create new execution context but share variable scope
- Loop variables persist after loop ends


15. EXAMPLE PROGRAM - BASIC
============================

Program.Main {
    # Declare variables
    Variables.New(Name)
    Variables.New(Age)
    
    # Get user input
    Variables.Name = Terminal.Ask("What's your name? ")
    Variables.Age = Terminal.Ask("What's your age? ")
    
    # Process with math
    Variables.New(NextAge)
    Variables.NextAge = Math.Eval(Variables.Age + 1)
    
    # Output with conditionals
    Terminal.Echo("Hello, " + Variables.Name + "!")
    
    Program.If(Math.Compare(Variables.Age, ">", 18)) {
        Terminal.Echo("You are an adult!")
    }
    
    Terminal.Echo("Next year you'll be " + Variables.NextAge)
    
    # Loops
    Program.Loop(3) {
        Terminal.Echo("Counting!")
    }
}


16. EXAMPLE PROGRAM - PROGRAM.ELSE
============================

Program.Main {
    Variables.New(Score)
    Variables.Score = Math.Random(0, 100)
    Terminal.Echo("Your score: " + Variables.Score)
    
    Program.If(Math.Compare(Variables.Score, ">", 50)) {
        Terminal.Echo("You passed!")
    }
    Program.Else {
        Terminal.Echo("You failed!")
    }
}


17. EXAMPLE PROGRAM - PROGRAM.NOT
============================

Program.Main {
    Variables.New(Username)
    Variables.Username = Terminal.Ask("Enter username: ")
    
    Program.If(Program.Not(Data.Compare(Variables.Username, "admin"))) {
        Terminal.Echo("Access denied - you are not an admin")
    }
    Program.Else {
        Terminal.Echo("Welcome, admin!")
    }
}


18. EXAMPLE PROGRAM - TIME AND MATH.RANDOM
============================

Program.Main {
    Terminal.Echo("Current time: " + Time.Current("hh:mm:ss tt"))
    
    Terminal.Echo("Processing...")
    Time.Wait(2000) # Wait 2 seconds
    Terminal.Echo("Done! Finished at " + Time.Current("HH:mm:ss"))
    
    # Random number game
    Variables.New(SecretNumber)
    Variables.SecretNumber = Math.Random(1, 10)
    Terminal.Echo("Secret number between 1 and 10 has been chosen")
    
    Variables.New(DiceRoll)
    Variables.DiceRoll = Math.Random(1, 6)
    Terminal.Echo("You rolled a " + Variables.DiceRoll)
}
